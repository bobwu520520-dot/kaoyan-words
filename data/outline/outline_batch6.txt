import json, re, collections
P = 'data/words.json'
d = json.load(open(P, encoding='utf-8'))
existing = collections.Counter(w['word'] for w in d['words'] if w.get('active') is not False)
added, bad, dups = [], [], 0
for ln in open('data/outline/outline_batch6.txt', encoding='utf-8'):
    ln = ln.strip()
    if not ln or ln.startswith(('EOF','python','wc')): continue
    parts = ln.split('|')
    if len(parts) < 3 or not parts[2].strip():
        bad.append(ln[:40]); continue
    word = parts[0].strip().lower()
    if not re.match(r'^[a-z][a-z-]+$', word):
        bad.append(ln[:40]); continue
    if existing[word] > 0:
        dups += 1; continue
    d['words'].append({
        'word': word, 'pos': parts[1].strip().split('/')[0].strip(), 'translation': parts[2].strip(),
        'tag': '大纲补充v10', 'tier': '高频重点', 'active': True, 'quality': 'outline-2026',
        'true_priority': '★★★★☆', 'exam_meaning': '', 'collocation_hint': '',
        'note': 'v10 考研高频补充词', 'studyEligible': True, 'secondary_meanings': '',
        'source': 'outline-v10', 'quality_score': 'B',
    })
    existing[word] += 1; added.append(word)
d['data_version'] = '5000-v10'
json.dump(d, open(P,'w',encoding='utf-8'), ensure_ascii=False, separators=(',',':'))
active = [w for w in d['words'] if w.get('active') is not False]
print('第六批新增:', len(added), '| 重复跳过:', dups, '| 坏行:', bad[:4])
print('有效词:', len(active), '| 分层:', dict(collections.Counter(w.get('tier') for w in active)))
